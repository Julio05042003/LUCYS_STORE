    <script>
        $(document).ready(function() {
            // Inicialización de Tablas Interactivas (DataTables)
            $('#tablaMasVendidos').DataTable({
                "pageLength": 5,
                "lengthMenu": [5, 10, 25],
                "language": { "url": "//cdn.datatables.net/plug-ins/1.13.6/i18n/es-ES.json" }
            });
            $('#tablaMenosVendidos').DataTable({
                "pageLength": 5,
                "lengthMenu": [5, 10, 25],
                "language": { "url": "//cdn.datatables.net/plug-ins/1.13.6/i18n/es-ES.json" }
            });
        });

        // Extraer y parsear los datos de forma segura
        const empleadosLabels = JSON.parse(document.getElementById('data-empleados-labels').textContent);
        const empleadosData = JSON.parse(document.getElementById('data-empleados-data').textContent);
        
        const categoriasLabels = JSON.parse(document.getElementById('data-categorias-labels').textContent);
        const categoriasValores = JSON.parse(document.getElementById('data-categorias-valores').textContent);

        // Renderizado de Gráfico de Barras - Empleados
        const ctxEmp = document.getElementById('chartEmpleados').getContext('2d');
        new Chart(ctxEmp, {
            type: 'bar',
            data: {
                labels: empleadosLabels,
                datasets: [{
                    label: 'Ventas ($)',
                    data: empleadosData,
                    backgroundColor: 'rgba(79, 70, 229, 0.6)',
                    borderColor: 'rgba(79, 70, 229, 1)',
                    borderWidth: 1
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                scales: {
                    y: { beginAtZero: true, grid: { color: 'rgba(255,255,255,0.1)' }, ticks: { color: '#fff' } },
                    x: { grid: { display: false }, ticks: { color: '#fff' } }
                },
                plugins: { legend: { display: false } }
            }
        });

        // Renderizado de Gráfico de Torta - Categorías
        const ctxCat = document.getElementById('chartCategorias').getContext('2d');
        new Chart(ctxCat, {
            type: 'doughnut',
            data: {
                labels: categoriasLabels,
                datasets: [{
                    data: categoriasValores,
                    backgroundColor: [
                        'rgba(16, 185, 129, 0.6)',
                        'rgba(245, 158, 11, 0.6)',
                        'rgba(59, 130, 246, 0.6)',
                        'rgba(239, 68, 68, 0.6)',
                        'rgba(139, 92, 246, 0.6)'
                    ],
                    borderWidth: 0
                }]
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                plugins: {
                    legend: { position: 'bottom', labels: { color: '#fff' } }
                }
            }
        });
    </script>
